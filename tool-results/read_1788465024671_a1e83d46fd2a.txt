    37→  uid: string
    38→  username: string | null
    39→  email: string
    40→  name: string | null
    41→  avatarUrl: string | null

    49→  side: string

    55→interface KycApplicationRow {
    56→  id: string
    57→  fullName: string
    58→  city: string
    59→  idType: string
    60→  status: string
    61→  rejectReason: string | null
    62→  submittedAt: string
    63→  reviewedAt: string | null
    64→  documents: KycDoc[]
    65→}
    66→
    67→interface UserDetail {
    68→  user: {
    69→    id: string; uid: string; username: string | null; email: string; name: string | null
    70→    avatarUrl: string | null; isBlocked: boolean; blockedReason: string | null
    71→    provider: string; country: string | null; phone: string | null; createdAt: string
    72→    // KYC summary (denormalized onto User)
    73→    kycStatus: string | null
    74→    kycSubmittedAt: string | null
    75→    kycApprovedAt: string | null
    76→    kycFullName: string | null
    77→    kycCity: string | null
    78→    kycIdType: string | null
    79→    kycRejectReason: string | null
    80→    // Whether a bcrypt password hash is set (Google-only users have none)
    81→    hasPassword: boolean
    82→  }
    83→  balances: { symbol: string; name: string; amount: number; usdValue: number; price: number; color: string; icon: string }[]
    84→  totalUsd: number
    85→  transactions: { id: string; type: string; token: string; amount: number; status: string; note: string | null; createdAt: string }[]
    86→  kycApplications: KycApplicationRow[]
    87→}

    91→const ID_LABELS: Record<string, string> = {

    96→
    97→export function UsersAdmin() {
    98→  const { toast } = useToast()
    99→  const [query, setQuery] = useState('')
   100→  const [results, setResults] = useState<SearchUser[]>([])
   101→  const [searching, setSearching] = useState(false)
   102→  const [selectedId, setSelectedId] = useState<string | null>(null)
   103→  const [detail, setDetail] = useState<UserDetail | null>(null)
   104→  const [loadingDetail, setLoadingDetail] = useState(false)
   105→  const [acting, setActing] = useState<string | null>(null)
   106→  // notify + reward + deduct dialogs
   107→  const [notifyOpen, setNotifyOpen] = useState(false)
   108→  const [rewardOpen, setRewardOpen] = useState(false)
   109→  const [deductOpen, setDeductOpen] = useState(false)
   110→  const [notifTitle, setNotifTitle] = useState('')
   111→  const [notifMsg, setNotifMsg] = useState('')
   112→  const [rewardToken, setRewardToken] = useState('USDT')
   113→  const [rewardAmount, setRewardAmount] = useState('')
   114→  const [rewardNote, setRewardNote] = useState('')
   115→  const [deductToken, setDeductToken] = useState('USDT')
   116→  const [deductAmount, setDeductAmount] = useState('')
   117→  const [deductNote, setDeductNote] = useState('')
   118→  const [cardDeductOpen, setCardDeductOpen] = useState(false)
   119→  const [cardDeductAmount, setCardDeductAmount] = useState('')
   120→  const [cardDeductNote, setCardDeductNote] = useState('')
   121→  const [warningOpen, setWarningOpen] = useState(false)
   122→  const [warningTitle, setWarningTitle] = useState('')
   123→  const [warningMsg, setWarningMsg] = useState('')
   124→
   125→  const search = useCallback(async (q: string) => {
   126→    if (!q.trim()) { setResults([]); return }
   127→    setSearching(true)
   128→    try {
   129→      const data = await apiFetch<{ users: SearchUser[] }>(`/api/admin/users/search?q=${encodeURIComponent(q.trim())}`)
   130→      setResults(data.users)
   131→    } catch (err: any) {
   132→      toast({ variant: 'destructive', title: 'Search failed', description: err.message })
   133→    } finally {
   134→      setSearching(false)
   135→    }
   136→  }, [toast])
   137→
   138→  useEffect(() => {
   139→    const id = setTimeout(() => search(query), 350)
   140→    return () => clearTimeout(id)
   141→  }, [query, search])
   142→
   143→  const loadDetail = useCallback(async (userId: string) => {
   144→    if (!getStoredToken()) return
   145→    setLoadingDetail(true)
   146→    try {
   147→      const data = await apiFetch<UserDetail>(`/api/admin/users/detail?userId=${userId}`)
   148→      setDetail(data)
   149→    } catch (err: any) {
   150→      // Silently skip auth errors (happens during logout)
   151→      const msg = String(err?.message || '')
   152→      if (msg.includes('401') || msg.toLowerCase().includes('unauthorized')) return
   153→      toast({ variant: 'destructive', title: 'Failed to load', description: err.message })
   154→    } finally {
   155→      setLoadingDetail(false)
   156→    }
   157→  }, [toast])
   158→
   159→  useEffect(() => {
   160→    if (selectedId) loadDetail(selectedId)
   161→  }, [selectedId, loadDetail])
   162→
   163→  async function act(kind: 'block' | 'unblock' | 'delete', extra?: { reason?: string }) {
   164→    if (!detail) return
   165→    setActing(kind)
   166→    try {
   167→      const body = kind === 'block' ? { userId: detail.user.id, reason: extra?.reason } : { userId: detail.user.id }
   168→      const res = await apiFetch<{ ok: boolean; message: string }>(`/api/admin/users/${kind}`, {
   169→        method: 'POST',
   170→        body: JSON.stringify(body),
   171→      })
   172→      toast({ title: kind === 'block' ? 'User Blocked' : kind === 'unblock' ? 'User Unblocked' : 'User Deleted', description: res.message })
   173→      if (kind === 'delete') {
   174→        setDetail(null)
   175→        setSelectedId(null)
   176→        setQuery('')
   177→        setResults([])
   178→      } else {
   179→        await loadDetail(detail.user.id)
   180→      }
   181→    } catch (err: any) {
   182→      toast({ variant: 'destructive', title: 'Action failed', description: err.message })
   183→    } finally {
   184→      setActing(null)
   185→    }
   186→  }
   187→
   188→  async function sendNotify() {
   189→    if (!detail) return
   190→    setActing('notify')
   191→    try {
   192→      const res = await apiFetch<{ ok: boolean; message: string }>(`/api/admin/users/notify`, {
   193→        method: 'POST',
   194→        body: JSON.stringify({ userId: detail.user.id, title: notifTitle, message: notifMsg }),
   195→      })
   196→      toast({ title: 'Notification sent', description: res.message })
   197→      setNotifyOpen(false); setNotifTitle(''); setNotifMsg('')
   198→    } catch (err: any) {
   199→      toast({ variant: 'destructive', title: 'Failed', description: err.message })
   200→    } finally {
   201→      setActing(null)
   202→    }
   203→  }
   204→
   205→  async function sendReward() {
   206→    if (!detail) return
   207→    setActing('reward')
   208→    try {
   209→      const res = await apiFetch<{ ok: boolean; message: string }>(`/api/admin/users/reward`, {
   210→        method: 'POST',
   211→        body: JSON.stringify({ userId: detail.user.id, token: rewardToken, amount: Number(rewardAmount), note: rewardNote }),
   212→      })
   213→      toast({ title: 'Reward sent 🎁', description: res.message })
   214→      setRewardOpen(false); setRewardAmount(''); setRewardNote('')
   215→      await loadDetail(detail.user.id)
   216→    } catch (err: any) {
   217→      toast({ variant: 'destructive', title: 'Failed', description: err.message })
   218→    } finally {
   219→      setActing(null)
   220→    }
   221→  }
   222→
   223→  async function sendDeduct() {
   224→    if (!detail) return
   225→    setActing('deduct')
   226→    try {
   227→      const res = await apiFetch<{ ok: boolean; message: string }>(`/api/admin/users/deduct`, {
   228→        method: 'POST',
   229→        body: JSON.stringify({ userId: detail.user.id, token: deductToken, amount: Number(deductAmount), note: deductNote }),
   230→      })
   231→      toast({ title: 'Funds deducted', description: res.message })
   232→      setDeductOpen(false); setDeductAmount(''); setDeductNote('')
   233→      await loadDetail(detail.user.id)
   234→    } catch (err: any) {
   235→      toast({ variant: 'destructive', title: 'Failed', description: err.message })
   236→    } finally {
   237→      setActing(null)
   238→    }
   239→  }
   240→
   241→  async function sendCardDeduct() {
   242→    if (!detail) return
   243→    setActing('cardDeduct')
   244→    try {
   245→      const res = await apiFetch<{ ok: boolean; message: string }>(`/api/admin/users/card-deduct`, {
   246→        method: 'POST',
   247→        body: JSON.stringify({ userId: detail.user.id, amount: Number(cardDeductAmount), note: cardDeductNote }),
   248→      })
   249→      toast({ title: 'Card funds deducted', description: res.message })
   250→      setCardDeductOpen(false); setCardDeductAmount(''); setCardDeductNote('')
   251→      await loadDetail(detail.user.id)
   252→    } catch (err: any) {
   253→      toast({ variant: 'destructive', title: 'Failed', description: err.message })
   254→    } finally {
   255→      setActing(null)
   256→    }
   257→  }
   258→
   259→  async function sendWarning() {
   260→    if (!detail) return
   261→    setActing('warning')
   262→    try {
   263→      const res = await apiFetch<{ ok: boolean; message: string }>(`/api/admin/users/warning`, {
   264→        method: 'POST',
   265→        body: JSON.stringify({ userId: detail.user.id, title: warningTitle, message: warningMsg }),
   266→      })
   267→      toast({ title: 'Warning sent ⚠️', description: res.message })
   268→      setWarningOpen(false); setWarningTitle(''); setWarningMsg('')
   269→    } catch (err: any) {
   270→      toast({ variant: 'destructive', title: 'Failed', description: err.message })
   271→    } finally {
   272→      setActing(null)
   273→    }
   274→  }
   275→
   276→  return (
   277→    <div className="space-y-5">
   278→      <div>
   279→        <h2 className="flex items-center gap-2 text-2xl font-extrabold tracking-tight">
   280→          <ShieldCheck className="h-6 w-6 text-gold" /> User Management
   281→        </h2>
   282→        <p className="text-sm text-muted-foreground">Search by username, UID, email, or name — then view balances, block, delete, notify, or reward.</p>
   283→      </div>
   284→
   285→      {/* Search bar */}
   286→      <div className="relative">
   287→        <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   288→        <Input
   289→          value={query}
   290→          onChange={(e) => setQuery(e.target.value)}
   291→          placeholder="Search username, UID, email, or name…"
   292→          className="border-border bg-card pl-10"
   293→        />
   294→        {searching && <Loader2 className="absolute right-3 top-1/2 h-4 w-4 -translate-y-1/2 animate-spin text-muted-foreground" />}
   295→      </div>
   296→
   297→      {/* Search results */}
   298→      {query.trim() && (
   299→        <div className="overflow-hidden glass-card rounded-2xl shadow-premium">
   300→          {results.length === 0 ? (
   301→            <div className="flex flex-col items-center justify-center py-12 text-center text-sm text-muted-foreground">
   302→              <Search className="mb-2 h-7 w-7 opacity-30" />
   303→              No users found
   304→            </div>
   305→          ) : (
   306→            results.map((u) => (
   307→              <button
   308→                key={u.id}
   309→                onClick={() => setSelectedId(u.id)}
   310→                className="flex w-full items-center gap-3 border-b border-border/50 px-4 py-3 text-left last:border-0 transition-colors hover:bg-secondary/40"
   311→              >
   312→                <div className="flex h-10 w-10 items-center justify-center rounded-full bg-gold/15 text-sm font-bold text-gold">
   313→                  {(u.username || u.email)[0].toUpperCase()}
   314→                </div>
   315→                <div className="flex-1 min-w-0">
   316→                  <div className="flex items-center gap-2">
   317→                    <span className="font-bold truncate">@{u.username || '—'}</span>
   318→                    {u.isBlocked && <span className="rounded bg-down/15 px-1.5 py-0.5 text-[9px] font-bold text-down">BLOCKED</span>}
   319→                  </div>
   320→                  <div className="truncate text-[11px] text-muted-foreground">{u.email} · UID {u.uid}</div>
   321→                </div>
   322→                <div className="text-right">
   323→                  <div className="font-mono text-sm font-semibold">{formatUsd(u.totalUsd)}</div>
   324→                  <div className="text-[10px] text-muted-foreground">{timeAgo(u.createdAt)}</div>
   325→                </div>
   326→              </button>
   327→            ))
   328→          )}
   329→        </div>
   330→      )}
   331→
   332→      {/* User detail drawer */}
   333→      <Sheet open={!!selectedId} onOpenChange={(v) => !v && (setSelectedId(null), setDetail(null))}>
   334→        <SheetContent className="w-full overflow-y-auto border-border bg-card sm:max-w-lg">
   335→          {loadingDetail || !detail ? (
   336→            <div className="flex h-full items-center justify-center">
   337→              <Loader2 className="h-8 w-8 animate-spin text-gold" />
   338→            </div>
   339→          ) : (
   340→            <>
   341→              <SheetHeader>
   342→                <SheetTitle className="flex items-center gap-2">
   343→                  User @{detail.user.username || '—'}
   344→                  {detail.user.isBlocked && <span className="rounded bg-down/15 px-1.5 py-0.5 text-[10px] font-bold text-down">BLOCKED</span>}
   345→                </SheetTitle>
   346→                <SheetDescription>UID {detail.user.uid} · joined {timeAgo(detail.user.createdAt)}</SheetDescription>
   347→              </SheetHeader>
   348→
   349→              <div className="mt-4 space-y-5">
   350→                {/* Profile picture + identity summary */}
   351→                <div className="flex flex-col items-center gap-2 rounded-xl border border-border bg-secondary/30 p-4 text-center">
   352→                  <VerifiedAvatar
   353→                    src={detail.user.avatarUrl}
   354→                    fallback={(detail.user.username || detail.user.email)[0].toUpperCase()}
   355→                    size="lg"
   356→                    verified={detail.user.kycStatus === 'approved'}
   357→                  />
   358→                  <div className="mt-1 text-base font-bold leading-tight">
   359→                    {detail.user.name || `@${detail.user.username || '—'}`}
   360→                  </div>
   361→                  <div className="flex flex-wrap items-center justify-center gap-1.5">
   362→                    <KycBadge status={detail.user.kycStatus} />
   363→                    {detail.user.isBlocked && (
   364→                      <span className="rounded bg-down/15 px-1.5 py-0.5 text-[9px] font-bold text-down">BLOCKED</span>
   365→                    )}
   366→                    <span className="rounded bg-secondary px-1.5 py-0.5 text-[9px] font-bold text-muted-foreground">
   367→                      {detail.user.provider === 'google' ? 'GOOGLE' : 'CREDENTIALS'}
   368→                    </span>
   369→                  </div>
   370→                </div>
   371→
   372→                {/* Email — prominent */}
   373→                <a
   374→                  href={`mailto:${detail.user.email}`}
   375→                  className="flex items-center gap-3 rounded-xl border border-gold/30 bg-gold/5 p-3 transition-colors hover:bg-gold/10"
   376→                >
   377→                  <Mail className="h-5 w-5 shrink-0 text-gold" />
   378→                  <div className="min-w-0 flex-1">
   379→                    <div className="text-[10px] uppercase tracking-wider text-muted-foreground">Email</div>
   380→                    <div className="truncate text-sm font-semibold text-foreground">{detail.user.email}</div>
   381→                  </div>
   382→                </a>
   383→
   384→                {/* Profile info */}
   385→                <div className="grid grid-cols-2 gap-2 rounded-xl border border-border bg-secondary/30 p-3 text-xs">
   386→                  <Info icon={AtSign} label="Username" value={`@${detail.user.username || '—'}`} />
   387→                  <Info icon={Hash} label="UID" value={detail.user.uid} />
   388→                  <Info icon={Mail} label="Phone" value={detail.user.phone || '—'} />
   389→                  <Info icon={Mail} label="Country" value={detail.user.country || '—'} />
   390→                </div>
   391→
   392→                {/* Password — never show plaintext, just note it's hashed */}
   393→                <div className="rounded-xl border border-border bg-secondary/20 p-3">
   394→                  <div className="flex items-center gap-2 text-xs">
   395→                    <KeyRound className="h-3.5 w-3.5 shrink-0 text-muted-foreground" />
   396→                    <div className="flex-1 min-w-0">
   397→                      <div className="text-[10px] uppercase tracking-wider text-muted-foreground">Password</div>
   398→                      {detail.user.hasPassword ? (
   399→                        <>
   400→                          <div className="font-mono text-sm font-semibold tracking-wider text-foreground">
   401→                            •••••••• <span className="text-[10px] font-normal text-muted-foreground">(hashed for security)</span>
   402→                          </div>
   403→                          <div className="mt-1 text-[10px] leading-snug text-muted-foreground">
   404→                            Stored as a one-way bcrypt hash. Plaintext is never saved and cannot be recovered — the user must reset it if forgotten.
   405→                          </div>
   406→                        </>
   407→                      ) : (
   408→                        <>
   409→                          <div className="text-sm font-semibold text-foreground">No password set</div>
   410→                          <div className="mt-1 text-[10px] leading-snug text-muted-foreground">
   411→                            This account signs in with Google only.
   412→                          </div>
   413→                        </>
   414→                      )}
   415→                    </div>
   416→                  </div>
   417→                </div>
   418→
   419→                {/* KYC section */}
   420→                <div className="rounded-xl border border-border bg-secondary/20 p-3">
   421→                  <div className="mb-2 flex items-center justify-between">
   422→                    <div className="flex items-center gap-1.5 text-xs font-bold uppercase tracking-wider text-muted-foreground">
   423→                      <ShieldCheck className="h-3.5 w-3.5" /> KYC Verification
   424→                    </div>
   425→                    <KycBadge status={detail.user.kycStatus} />
   426→                  </div>
   427→                  {detail.user.kycStatus && detail.user.kycStatus !== 'none' ? (
   428→                    <div className="grid grid-cols-2 gap-2 text-xs">
   429→                      <Info icon={UserIcon} label="Full Name" value={detail.user.kycFullName || '—'} />
   430→                      <Info icon={MapPin} label="City" value={detail.user.kycCity || '—'} />
   431→                      <Info icon={IdCard} label="ID Type" value={ID_LABELS[detail.user.kycIdType || ''] || detail.user.kycIdType || '—'} />
   432→                      <Info icon={Clock} label="Submitted" value={detail.user.kycSubmittedAt ? new Date(detail.user.kycSubmittedAt).toLocaleDateString() : '—'} />
   433→                    </div>
   434→                  ) : (
   435→                    <div className="text-[11px] text-muted-foreground">User has not submitted KYC verification.</div>
   436→                  )}
   437→
   438→                  {detail.user.kycStatus === 'rejected' && detail.user.kycRejectReason && (
   439→                    <div className="mt-2 flex items-start gap-2 rounded-lg border border-down/40 bg-down/5 p-2 text-[11px] text-down">
   440→                      <AlertTriangle className="mt-0.5 h-3 w-3 shrink-0" />
   441→                      <span>Reject reason: {detail.user.kycRejectReason}</span>
   442→                    </div>
   443→                  )}
   444→
   445→                  {/* KYC documents (front + back) — only visible if they still exist in DB */}
   446→                  {detail.kycApplications.length > 0 && (
   447→                    <div className="mt-3 space-y-3">
   448→                      {detail.kycApplications.map((app) => (
   449→                        <div key={app.id} className="rounded-lg border border-border/60 bg-secondary/30 p-2">
   450→                          <div className="mb-1.5 flex items-center justify-between text-[10px] text-muted-foreground">
   451→                            <span>Application · {app.status}</span>
   452→                            <span>{new Date(app.submittedAt).toLocaleDateString()}</span>
   453→                          </div>
   454→                          {app.documents.length > 0 ? (
   455→                            <div className="grid grid-cols-1 gap-2">
   456→                              {app.documents.map((doc) => (
   457→                                <div key={doc.id} className="space-y-1">
   458→                                  <div className="flex items-center justify-between">
   459→                                    <span className="text-[10px] font-bold uppercase tracking-wider text-gold">
   460→                                      {doc.side === 'back' ? 'Back of ID' : 'Front of ID'}
   461→                                    </span>
   462→                                    {doc.deleteAfter && (
   463→                                      <span className="text-[9px] text-muted-foreground">
   464→                                        Auto-deletes {new Date(doc.deleteAfter).toLocaleDateString()}
   465→                                      </span>
   466→                                    )}
   467→                                  </div>
   468→                                  <div className="rounded border border-border overflow-hidden">
   469→                                    <img
   470→                                      src={`/api/kyc/document?id=${doc.id}`}
   471→                                      alt={`${doc.side} of ID`}
   472→                                      className="w-full max-h-[180px] object-contain bg-black/30"
   473→                                    />
   474→                                  </div>
   475→                                  <div className="flex gap-1.5">
   476→                                    <a
   477→                                      href={`/api/kyc/document?id=${doc.id}&download=true`}
   478→                                      target="_blank"
   479→                                      rel="noreferrer"

   746→/** Colored pill showing the user's KYC status. */
   747→function KycBadge({ status }: { status: string | null }) {
   748→  const map: Record<string, { label: string; cls: string }> = {
   749→    approved: { label: 'KYC APPROVED', cls: 'bg-up/15 text-up' },
   750→    pending: { label: 'KYC PENDING', cls: 'bg-gold/15 text-gold' },
   751→    rejected: { label: 'KYC REJECTED', cls: 'bg-down/15 text-down' },
   752→    none: { label: 'NO KYC', cls: 'bg-secondary text-muted-foreground' },
   753→  }
   754→  const s = map[status || 'none'] || map.none
   755→  return <span className={`rounded px-1.5 py-0.5 text-[9px] font-bold ${s.cls}`}>{s.label}</span>
   756→}
